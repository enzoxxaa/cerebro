\section{El problema}

\begin{frame}{Poblaciones = pesos de Boltzmann}
  \begin{block}{La única definición que necesitamos}
    $$\frac{P_A}{P_B} = e^{-\Delta G_{AB}/RT} \qquad\Longleftrightarrow\qquad \Delta G_{AB} = -RT\ln\frac{P_A}{P_B}$$
  \end{block}
  \vspace{0.5em}
  \begin{itemize}
    \item A 300 K: $RT \approx \SI{2.5}{\kilo\joule\per\mole}$
    \item Regla práctica: \alert{un factor $e$ en población cuesta 2.5 kJ/mol}
    \item Un factor 10 cuesta $RT\ln 10 \approx \SI{5.7}{\kilo\joule\per\mole}$
  \end{itemize}
  \vspace{0.5em}
  \Fuente{Sin matices: es la definición del ensemble canónico.}
\end{frame}

\begin{frame}{¿Por qué la MD directa no basta?}
  \begin{block}{Teoría del estado de transición}
    $$k = \kappa\,\frac{k_BT}{h}\,e^{-\Delta G^{\ddagger}/RT}, \qquad \tau = 1/k$$
  \end{block}
  \begin{center}
  \begin{tabular}{ccc}
    \toprule
    $\Delta G^{\ddagger}$ & $\tau$ (estimado) & ¿cruza en 100 ns? \\
    \midrule
    20 kJ/mol & $\sim$30 ns & sí, varias veces \\
    \alert{$\sim$22 kJ/mol} & \alert{$\sim$100 ns} & \alert{frontera} \\
    40 kJ/mol & $\sim$90–150 $\mu$s & \textbf{no} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.8em}
  \textbf{100 ns de MD compran barreras de hasta $\sim$22 kJ/mol.} Las transiciones de lazos CDR viven en 25–45 kJ/mol.
\end{frame}

\begin{frame}{$\Delta G$ y $\Delta G^{\ddagger}$ son independientes — la distinción que sostiene todo}
  Por balance detallado (tasas escritas desde el mismo estado de transición, dividiendo — $F(s^\ddagger)$ se cancela):
  $$\frac{k_{A\to B}}{k_{B\to A}} = e^{-\Delta G_{AB}/RT} = \frac{P_B}{P_A}$$
  \begin{itemize}
    \item \alert{La barrera no aparece en el cociente.} Controla \textbf{cuánto tarda el sistema} en equilibrar, no \textbf{dónde} está el equilibrio
    \item Este sistema: pozos someros ($\Delta G\approx6$ kJ/mol, $2.4RT$) + muros altos ($\Delta G^{\ddagger}\approx25$--45 kJ/mol, $10$--$18RT$)
    \item Consecuencia directa: alta precisión exigida sobre un número pequeño, en un sistema que \textbf{se niega a equilibrarse solo}
  \end{itemize}
  \vspace{0.4em}
  \Fuente{El balance detallado reaparece dos veces más en esta charla — al simetrizar $C(\tau)$ en tICA, y al exigir reversibilidad en el MSM. Es la misma condición física, no una coincidencia notacional.}
\end{frame}

\begin{frame}{El resultado que hay que poder explicar}
  \centering
  \includegraphics[width=0.85\textwidth]{fig4-msm-poblaciones.png}
  \Pie{Fig. 4 del paper: paisajes con probabilidades de macroestado y tiempos de transición, parent E06 vs. huE06 v1.1 (izq.) y vs. huE06 v1.4 (der.). Rombo blanco = estructura cristalográfica.}
  \vspace{0.3em}

  {\scriptsize E06: \textbf{92.6\%} / 7.4\% \quad$\to$\quad huE06 v1.1: 16.3\% / 48.4\% / 35.3\%}

  \alert{$\Delta\Delta G \approx 10$ kJ/mol} — comparable al error típico de un campo de fuerza
\end{frame}

\begin{frame}{¿Y si el muestreo está mal repartido?}
  Si se agrupan trayectorias con pesos $w_i$ distintos de los pesos de Boltzmann verdaderos $\pi_i$:
  $$\hat F(s) = F(s) - RT\ln\frac{w_i}{\pi_i}, \qquad s\in \text{cuenca } i$$
  \begin{itemize}
    \item La \textbf{forma} de cada pozo sale correcta (el error no depende de $s$ dentro de la cuenca)
    \item Las \textbf{profundidades relativas} —es decir, las poblaciones— salen mal
    \item Y el error \textbf{no decrece} con más muestreo: es sesgo sistemático, no ruido
  \end{itemize}
  \vspace{0.5em}
  \Fuente{Se retoma al final: de aquí sale por qué ninguna etapa del pipeline puede saltarse la reponderación.}
\end{frame}

\begin{frame}{Conclusión del Bloque 1}
  \begin{center}
    \Large
    Hace falta cruzar barreras de $\sim$40 kJ/mol\\
    sin esperar $\sim$100 $\mu$s por cruce\\[0.6em]
    \alert{sin destruir las poblaciones al hacerlo.}
  \end{center}
\end{frame}

\section{Inventar la metadinámica}

\begin{frame}{¿Qué palancas existen?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Temperatura global (REMD)}
      \begin{itemize}
        \item[\textcolor{minaranja}{$-$}] acelera \emph{todo} indiscriminadamente
        \item[\textcolor{minaranja}{$-$}] reponderar a 300 K es caro (varianza $\sim e^{\sqrt{N}}$)
        \item[\textcolor{minaranja}{$-$}] no se puede apuntar a un lazo
        \item[\textcolor{teal}{$+$}] no requiere elegir CVs
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Sesgar el potencial sobre una CV}
      \begin{itemize}
        \item[\textcolor{teal}{$+$}] selectivo — solo empuja donde uno decide
        \item[\textcolor{teal}{$+$}] el sesgo se conoce exactamente: barato de quitar
        \item[\textcolor{minaranja}{$-$}] hay que \alert{elegir} las CVs de antemano
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1em}
  \centering \alert{\large Metadinámica: sesgar sobre la CV.}
\end{frame}

\begin{frame}{Un sesgo sobre la CV se suma al paisaje}
  Si $U_V(\mathbf{r}) = U(\mathbf{r}) + V(\xi(\mathbf{r}))$, entonces \textbf{exactamente}:
  $$\boxed{F_V(s) = F(s) + V(s)}$$
  \vspace{0.4em}
  \begin{itemize}
    \item Se cumple porque $V$ depende de $\mathbf{r}$ \emph{solo a través de} $s=\xi(\mathbf{r})$
    \item El sesgo que aplana el paisaje: $F_V = 0 \iff V(s) = -F(s)$
  \end{itemize}
  \begin{alertblock}{El problema}
    Para construir $V=-F$ hace falta conocer $F$ — que es justo lo que queremos medir. \textbf{Huevo y gallina.}
  \end{alertblock}
\end{frame}

\begin{frame}{La solución: sesgo adaptativo}
  Depositar, cada $\tau_G$ pasos, una gaussiana donde está el sistema \emph{ahora}:
  $$V(s,t) = \sum_{\substack{t'=\tau_G,2\tau_G,\dots \\ t' < t}} W\, e^{-(s-\xi(\mathbf{r}(t')))^2/2\sigma^2}$$
  \vspace{0.2em}
  {\scriptsize\color{migris} Una gaussiana por cada instante de depósito ya pasado — la suma \textbf{crece}, nunca se borra.}
  \vspace{0.3em}
  \begin{itemize}
    \item \alert{En tiempo real}: PLUMED acoplado a GROMACS calcula la fuerza de $U+V$ en \textbf{cada paso} (2 fs), no solo al depositar — el sistema \textbf{siente} el sesgo acumulado hasta ese instante
    \item El sistema \textbf{va dejando arena donde pisa}
    \item Bucle de realimentación: $\dfrac{\partial V}{\partial t} \propto e^{-\beta[F(s)+V(s,t)]}$ — pozo profundo $\to$ más tiempo ahí $\to$ deposita rápido $\to$ $F+V$ sube ahí $\to$ atrae menos $\to$ se autolimita
  \end{itemize}
  \begin{alertblock}{Pero...}
    La deposición \textbf{nunca para}. El sistema acaba explorando regiones absurdas (\emph{overfilling}).
  \end{alertblock}
\end{frame}

\begin{frame}{El sesgo como molde negativo del paisaje}
  \centering
  \resizebox{\textwidth}{!}{\MoldeNegativoFigure[1]}
  \vspace{0.4em}
  \small\color{migris} $t_1<t_2<t_3$: \textbf{tres instantes de la misma corrida} — no tres simulaciones distintas. En $A$ y $B$ se siguen apilando gaussianas porque el sistema los sigue visitando; para $t_3$ ya casi igualan $-F(s)$ (curva gris de fondo, panel 2).
\end{frame}

\begin{frame}{Well-tempered: frenar la deposición}
  La altura decae donde ya hay sesgo \alert{acumulado} — no donde hay más energía: $W(s,t) = W_0\, e^{-V(s,t)/k_B\Delta T}$
  \vspace{0.3em}
  \begin{block}{Convergencia}
    $$V(s) \xrightarrow[t\to\infty]{} -\frac{\gamma-1}{\gamma}F(s), \qquad \gamma \equiv \frac{T+\Delta T}{T}$$
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Barrera residual: $F_V = F/\gamma$
        \item Muestreo: $p_V \propto \peq^{\,1/\gamma}$
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Recuperar $F$: $F = -\dfrac{\gamma}{\gamma-1}V$
        \item $\gamma\to 1$: MD normal. $\gamma\to\infty$: metaD estándar
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{El precio}

\begin{frame}{La cinética queda comprimida, no solo cambiada}
  Factor de aceleración por transición: $\alpha = e^{\beta\Delta G^{\ddagger}(\gamma-1)/\gamma}$ — \alert{depende de la barrera}
  $$\frac{\tau_A^V}{\tau_B^V} = \left(\frac{\tau_A}{\tau_B}\right)^{1/\gamma}$$
  \vspace{0.3em}
  \begin{exampleblock}{Ejemplo}
    Si en el sistema real $A$ tarda \textbf{23\,000$\times$} más que $B$, sesgado con $\gamma=10$:
    $$23\,000^{1/10} \approx \mathbf{2.7}$$
  \end{exampleblock}
  \textbf{Cuatro órdenes de magnitud, aplastados a un factor 3.} No es un efecto secundario — es el mecanismo.
\end{frame}

\begin{frame}{¿Se puede recuperar la cinética real?}
  Sí — \textbf{metadinámica infrecuente} (Tiwary \& Parrinello, \textit{PRL} \textbf{111}:230602, 2013; 353 citas)
  \begin{itemize}
    \item Depositar sesgo \textbf{muy espaciado}: casi no contamina el estado de transición
    \item El factor de aceleración se \textbf{mide}, $\alpha=\langle e^{\beta V(s,t)}\rangle$, no se asume
  \end{itemize}
  \begin{alertblock}{Régimen opuesto al del paper}
    Aquí: cada 2 ps, 500\,000 gaussianas en 1 $\mu$s — diseñado para \textbf{explorar}, no para medir tasas.
  \end{alertblock}
  \centering \alert{El paper no necesita recuperar cinética: necesita estructuras.}
\end{frame}

\begin{frame}{Las CVs son una elección}
  La fuerza de sesgo sobre el átomo $i$: $\mathbf{f}_i = -\dfrac{dV}{ds}\,\nabla_{\mathbf{r}_i}\xi(\mathbf{r})$
  \begin{itemize}
    \item Si el átomo $i$ no define la CV: $\nabla_{\mathbf{r}_i}\xi = \mathbf{0}$
    \item El sesgo empuja \textbf{literalmente} solo los átomos de CDR1/CDR3 ($\sim$0.3\% del sistema)
    \item Todo grado de libertad ortogonal (p. ej. HV2) recibe \alert{ayuda cero}
  \end{itemize}
  \vspace{0.5em}
  \Fuente{No es una sutileza — es una restricción mecánica del método.}
\end{frame}

\begin{frame}{La jugada: tirar $F$ y la cinética}
  \begin{center}
  \begin{tabular}{lcc}
    \toprule
    Producto & Estado & ¿Se usa? \\
    \midrule
    Energía libre $F(s)$ & sin convergencia verificada & \textcolor{minaranja}{no} \\
    Cinética & comprimida, no física & \textcolor{minaranja}{no} \\
    \textbf{Conjunto de estructuras} & diverso y útil & \textcolor{teal}{\textbf{sí}} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.8em}
  \begin{quote}
    \small ``Being only used to generate initial conformations, the implementation of metadynamics can be simple and fast.''\\
    \Fuente{Biswas, Lickert \& Stock, \textit{J. Phys. Chem. B} \textbf{122}:5508 (2018)}
  \end{quote}
  \centering \alert{La convergencia deja de ser un requisito.}
\end{frame}

\section{MD sembrada}

\begin{frame}{De estructuras a dinámica sin sesgo}
  \begin{itemize}
    \item Las trayectorias de metadinámica se agrupan: \textbf{clustering jerárquico} (\textit{average linkage}, cpptraj), corte \textbf{1.3 \r{A}}
    \item Cada representante de cluster $\to$ \textbf{100 ns de MD sin sesgo} (AMBER 20, \texttt{pmemd.cuda})
    \item El paper reporta ``\textit{a large number of clusters}'' — el número exacto \alert{no se publica}
  \end{itemize}

  \begin{alertblock}{Lo que esto fija, sin que nadie lo controle explícitamente}
    El número de clusters por variante determina $w_i$ — y con él, cuántas semillas hay por región del paisaje.
  \end{alertblock}
\end{frame}

\section{tICA}

\begin{frame}{Reducir dimensión: ¿pero cuál?}
  Cientos de trayectorias de 100 ns, en un espacio de muchas torsiones. Hace falta proyectar a 2D para poder mirar el paisaje.
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{PCA}
      \begin{itemize}
        \item Maximiza \textbf{varianza}
        \item[\textcolor{minaranja}{$-$}] un lazo puede bambolearse mucho sin ser el modo \emph{lento}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{tICA}
      \begin{itemize}
        \item Maximiza \textbf{autocorrelación} a un lag $\tau$
        \item[\textcolor{teal}{$+$}] encuentra lo \textbf{lento}, no lo amplio
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{tICA: el problema de autovalores}
  $$C(\tau)\,v = \lambda\, C(0)\,v$$
  \begin{itemize}
    \item $C(\tau)$: covarianza \textbf{desfasada} en el tiempo. $C(0)$: covarianza instantánea
    \item PCA solo usa $C(0)$ — tICA usa \textbf{ambas}
    \item Los autovectores de mayor autovalor son las direcciones que \textbf{menos se decorrelacionan} al esperar $\tau$: las lentas
  \end{itemize}
  \vspace{0.6em}
  \Fuente{Pérez-Hernández \& Noé, \textit{J. Chem. Theory Comput.} \textbf{12}:6118 (2016) — tICA jerárquico, usado por PyEMMA 2.}
\end{frame}

\begin{frame}{Las features del paper}
  \begin{itemize}
    \item Torsiones de \emph{backbone} de \textbf{CDR1, CDR3 y HV2}
    \item Lag de tICA: \textbf{10 ns} (distinto del lag del MSM, 15 ns)
    \item Software: PyEMMA 2
  \end{itemize}
  \vspace{0.6em}
  \begin{alertblock}{La asimetría ya establecida en el Bloque 3}
    HV2 \textbf{nunca fue una CV de metadinámica} — pero sí es una \textbf{feature de tICA}. Se mide, aunque no se muestreó con sesgo.
  \end{alertblock}
\end{frame}

\section{El Markov State Model}

\begin{frame}{Lo que un MSM estima, sin álgebra de más}
  \begin{itemize}
    \item Partición en \textbf{100 microestados} (k-means sobre las coordenadas tICA)
    \item Se cuentan transiciones $i\to j$ a un \textbf{lag} $\tau=15$ ns; normalizando, la \textbf{matriz de transición} $T(\tau)$ — es el estimador de \textbf{máxima verosimilitud} bajo la hipótesis de Markov, no una receta ad hoc
    \item De $T(\tau)$, por álgebra lineal, salen \textbf{tres} cosas: poblaciones de equilibrio $\pi$, tiempos característicos $t_i=-\tau/\ln\lambda_i$, y — vía \textbf{PCCA+} (Röblitz \& Weber, 2013) — macroestados agrupando los 100 microestados
  \end{itemize}
  \vspace{0.6em}
  \centering \alert{\large Un solo objeto, $T(\tau)$. Todo lo demás es consecuencia, no un ajuste aparte.}
\end{frame}

\begin{frame}{El nodo que legitima todo el pipeline}
  \begin{center}
    \Large
    \alert{$\pi$ es independiente de dónde se sembró.}
  \end{center}
  \vspace{0.8em}
  \begin{itemize}
    \item $\pi$ sale de las \textbf{tasas locales} $T_{ij}$, no de cuántas veces se visitó cada microestado
    \item Sembrar más en una región mejora la \textbf{precisión} de $T_{ij}$ ahí — no cambia su valor esperado
    \item \textbf{Esto resuelve} el problema planteado en el Bloque 1: $\hat F = F - RT\ln(w_i/\pi_i)$ — el MSM reponde $w_i\to\pi_i$
  \end{itemize}
  \vspace{0.5em}
  \Fuente{Es la respuesta formal a ``pero la siembra estaba sesgada''.}
\end{frame}

\begin{frame}{Validación: ¿es válido el modelo?}
  \begin{itemize}
    \item \textbf{Test de Chapman--Kolmogorov}: $T(n\tau) \approx [T(\tau)]^n$ — si falla, el lag es demasiado corto (no markoviano)
    \item \textbf{VAMP} (Wu \& Noé, 2017): score variacional para elegir lag y features óptimos
    \item Fracción de estados \textbf{conectados} — necesaria para que $\pi$ tenga sentido
  \end{itemize}
  \vspace{0.6em}
  \Fuente{Prinz \textit{et al.}, \textit{J. Chem. Phys.} \textbf{134}:174105 (2011) — la referencia canónica del test CK aplicado a MSMs.}
\end{frame}

\begin{frame}{A2 — la versión mecánica}

    Los átomos de HV2 no recibieron\\

    en todo el microsegundo de metadinámica.
  \end{center}

  \small\color{migris}

\begin{frame}{A2 — las implicancias, con nombre y apellido}
  \textbf{La estructura lógica exacta del problema:}
  \begin{enumerate}
    \setlength{\itemsep}{0.5em}
    \item HV2 \textbf{nunca} fue CV $\Rightarrow$ fuerza de sesgo cero sobre sus átomos (Bloque 3)
    \item HV2 \textbf{sí} es \textit{feature} de tICA y \textbf{sí} entra al MSM (Bloque \textit{tICA})
    \item El paper atribuye la pérdida de afinidad a \alert{``\textit{missing interactions with the HV2 loop}''} (Fig. 5, Discusión)
  \end{enumerate}
  \vspace{0.5em}
  \begin{alertblock}{Las dos lecturas que el paper no distingue}
    \textbf{(a)} el movimiento de HV2 es una \emph{respuesta correlacionada}, real, transmitida por física clásica al reacomodarse CDR1/CDR3 — legítimo \emph{si} está bien muestreado\\
    \textbf{(b)} es un \emph{artefacto de submuestreo}: ningún método, en ninguna etapa posterior del análisis (ni tICA ni el MSM), puede rescatar un cruce de barrera de HV2 que jamás se sesgó
  \end{alertblock}
  \centering \alert{El paper no ofrece ninguna forma de distinguir (a) de (b) — y la conclusión mecanicista depende de que sea (a).}
\end{frame}
